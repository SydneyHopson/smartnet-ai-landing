import { z } from "zod";

export const conversationRoleSchema = z.enum([
  "system",
  "assistant",
  "user",

  id: z.string(),

  sessionId: z.string(),

  status: z
    .enum([
      "not_started",
      "collecting_scope",
      "awaiting_customer",
      "ready_for_pricing",
      "completed",
      "error",
    ])
    .default("not_started"),

  messages: z.array(conversationMessageSchema).default([]),

  currentQuestion: z.string().nullable().default(null),

  currentQuestionKey: z.string().nullable().default(null),

  unansweredQuestionKeys: z.array(z.string()).default([]),

  completedQuestionKeys: z.array(z.string()).default([]),

  lastCustomerMessage: z.string().nullable().default(null),

  lastAssistantMessage: z.string().nullable().default(null),

  readyForPricing: z.boolean().default(false),

  confidenceScore: z.number().min(0).max(100).default(0),

  createdAt: z.string().datetime(),

  updatedAt: z.string().datetime(),
});

export type ConversationMessage = z.infer<
  typeof conversationMessageSchema
>;

export type EstimatorConversationState = z.infer<
  typeof estimatorConversationStateSchema
>;

export function createEstimatorConversationState(
  sessionId: string
): EstimatorConversationState {
  const now = new Date().toISOString();

  return estimatorConversationStateSchema.parse({
    sessionId,
    status: "not_started",
    messages: [],
    currentQuestion: null,
    currentQuestionKey: null,
    unansweredQuestionKeys: [],
    completedQuestionKeys: [],
    lastCustomerMessage: null,
    lastAssistantMessage: null,
    readyForPricing: false,
    confidenceScore: 0,
    createdAt: now,
    updatedAt: now,
  });
}

import type { ProjectEstimate } from "../domain/project-estimate";

export type EstimatorQuestionPriority =
  | "critical"
  | "high"
  | "normal"
  | "optional";

export type EstimatorQuestion = {
  key: string;
  question: string;
  reason: string;
  priority: EstimatorQuestionPriority;
  category:
    | "project"
    | "property"
    | "cameras"
    | "network"
    | "wifi"
    | "access_control"
    | "cabling"
    | "installation"
    | "commercial";
};

const priorityOrder: Record<EstimatorQuestionPriority, number> = {
  critical: 0,
  high: 1,
  normal: 2,
  optional: 3,
};

function hasKnownQuantity(quantity: {
  value: number | null;
}): boolean {
  return quantity.value !== null;
}

function addQuestion(
  questions: EstimatorQuestion[],
  question: EstimatorQuestion
): void {
  const alreadyAdded = questions.some(
    (existingQuestion) => existingQuestion.key === question.key
  );

  if (!alreadyAdded) {
    questions.push(question);
  }
}

export function getEstimatorQuestions(
  project: ProjectEstimate
): EstimatorQuestion[] {
  const questions: EstimatorQuestion[] = [];

  /*
   * ---------------------------------------------------------
   * Core project questions
   * ---------------------------------------------------------
   */

  if (!project.customerIntent.summary.trim()) {
    addQuestion(questions, {
      key: "customerIntent.summary",
      question:
        "Tell me what you want the finished system to accomplish and what problems you are experiencing today.",
      reason:
        "The estimator needs to understand the customer's goals before recommending equipment.",
      priority: "critical",
      category: "project",
    });
  }

  if (!project.property.projectType) {
    addQuestion(questions, {
      key: "property.projectType",
      question:
        "What type of property is this, such as a home, office, retail store, warehouse, medical facility, school, church, or something else?",
      reason:
        "Property type changes the equipment, installation method, labor assumptions, and follow-up questions.",
      priority: "critical",
      category: "property",
    });
  }

  if (!hasKnownQuantity(project.property.squareFootage)) {
    addQuestion(questions, {
      key: "property.squareFootage",
      question:
        "Approximately how many square feet are included in this project?",
      reason:
        "Square footage helps estimate device density, cable distance, labor, and project duration.",
      priority: "critical",
      category: "property",
    });
  }

  if (!hasKnownQuantity(project.property.numberOfFloors)) {
    addQuestion(questions, {
      key: "property.numberOfFloors",
      question:
        "How many floors or separate levels are included in the project?",
      reason:
        "Multiple floors can require additional pathways, network closets, riser cabling, and labor.",
      priority: "high",
      category: "property",
    });
  }

  if (project.property.constructionType === "unknown") {
    addQuestion(questions, {
      key: "property.constructionType",
      question:
        "Is this an existing finished building, an unfinished space, a renovation, or new construction?",
      reason:
        "Construction phase strongly affects cable access, wall repair, pathways, labor, and scheduling.",
      priority: "critical",
      category: "property",
    });
  }

  if (project.property.ceilingType === "unknown") {
    addQuestion(questions, {
      key: "property.ceilingType",
      question:
        "What kind of ceilings are present: drywall, drop ceiling, open ceiling, warehouse deck, or a mixture?",
      reason:
        "Ceiling construction affects cable pathways, mounting methods, lift requirements, and installation time.",
      priority: "high",
      category: "property",
    });
  }

  if (!hasKnownQuantity(project.property.ceilingHeightFeet)) {
    addQuestion(questions, {
      key: "property.ceilingHeightFeet",
      question:
        "Approximately how high are the ceilings in the areas where equipment will be installed?",
      reason:
        "Mounting height determines ladder or lift needs and changes installation labor.",
      priority: "high",
      category: "property",
    });
  }

  if (project.property.occupiedDuringInstall === null) {
    addQuestion(questions, {
      key: "property.occupiedDuringInstall",
      question:
        "Will the building remain occupied and operating while the installation is being performed?",
      reason:
        "An occupied installation can require phased work, after-hours labor, noise controls, and additional protection.",
      priority: "normal",
      category: "installation",
    });
  }

  /*
   * ---------------------------------------------------------
   * Camera scope
   * ---------------------------------------------------------
   */

  if (project.cameras.requested) {
    if (
      !hasKnownQuantity(project.cameras.interiorCount) &&
      !hasKnownQuantity(project.cameras.exteriorCount)
    ) {
      addQuestion(questions, {
        key: "cameras.locationCounts",
        question:
          "Do you need interior cameras, exterior cameras, or both, and approximately how many locations should be covered?",
        reason:
          "Interior and exterior cameras require different hardware, mounts, weather protection, and labor.",
        priority: "critical",
        category: "cameras",
      });
    }

    if (project.cameras.coverageGoals.length === 0) {
      addQuestion(questions, {
        key: "cameras.coverageGoals",
        question:
          "Which areas must be covered, such as entrances, parking, loading docks, hallways, registers, inventory, or the full perimeter?",
        reason:
          "Coverage goals determine camera quantity, lens type, resolution, placement, and storage requirements.",
        priority: "critical",
        category: "cameras",
      });
    }

    if (!hasKnownQuantity(project.cameras.recordingDays)) {
      addQuestion(questions, {
        key: "cameras.recordingDays",
        question:
          "How many days of recorded video would you like the system to retain?",
        reason:
          "Retention requirements determine recorder and storage capacity.",
        priority: "high",
        category: "cameras",
      });
    }

    if (project.cameras.existingSystem === null) {
      addQuestion(questions, {
        key: "cameras.existingSystem",
        question:
          "Is there an existing camera or recording system that must be reused, replaced, or integrated?",
        reason:
          "Existing equipment may affect compatibility, licensing, demolition, and migration labor.",
        priority: "high",
        category: "cameras",
      });
    }

    if (project.cameras.remoteViewingRequired === null) {
      addQuestion(questions, {
        key: "cameras.remoteViewingRequired",
        question:
          "Do you need remote viewing from phones or computers, and how many users will need access?",
        reason:
          "Remote access affects system selection, networking, permissions, and configuration time.",
        priority: "normal",
        category: "cameras",
      });
    }

    if (
      hasKnownQuantity(project.cameras.exteriorCount) &&
      (project.cameras.exteriorCount.value ?? 0) > 0 &&
      project.cameras.mountingSurfaces.length === 0
    ) {
      addQuestion(questions, {
        key: "cameras.mountingSurfaces",
        question:
          "What surfaces will the exterior cameras mount to, such as brick, stucco, siding, metal, soffit, poles, or concrete?",
        reason:
          "Mounting surfaces determine hardware, weatherproofing, tools, pathways, and installation difficulty.",
        priority: "high",
        category: "cameras",
      });
    }
  }

  /*
   * ---------------------------------------------------------
   * Network and rack scope
   * ---------------------------------------------------------
   */

  if (
    project.network.requested ||
    project.wifi.requested ||
    project.cameras.requested ||
    project.accessControl.requested
  ) {
    if (!project.network.internetProvider) {
      addQuestion(questions, {
        key: "network.internetProvider",
        question:
          "Who is the current internet provider, and where are the modem and router located?",
        reason:
          "The current internet handoff and equipment location affect network design and rack placement.",
        priority: "normal",
        category: "network",
      });
    }

    if (project.network.existingRouter === null) {
      addQuestion(questions, {
        key: "network.existingRouter",
        question:
          "Do you already have a router or firewall that should remain in the new system?",
        reason:
          "Existing routing equipment can affect compatibility, configuration, licensing, and replacement costs.",
        priority: "high",
        category: "network",
      });
    }

    if (project.network.existingSwitches === null) {
      addQuestion(questions, {
        key: "network.existingSwitches",
        question:
          "Are there existing network or PoE switches that should be reused?",
        reason:
          "Switch capacity, PoE budget, speed, and available ports directly affect equipment cost.",
        priority: "high",
        category: "network",
      });
    }

    if (project.network.existingRack === null) {
      addQuestion(questions, {
        key: "network.existingRack",
        question:
          "Is there an existing network rack, cabinet, or structured wiring panel?",
        reason:
          "The estimator must determine whether rack hardware, mounting, power, cooling, and cleanup are required.",
        priority: "high",
        category: "network",
      });
    }

    if (
      project.network.existingRack === true &&
      !project.network.rackLocation
    ) {
      addQuestion(questions, {
        key: "network.rackLocation",
        question:
          "Where is the existing rack or network equipment located?",
        reason:
          "Rack location affects cable distance, pathway design, access, power, and environmental requirements.",
        priority: "high",
        category: "network",
      });
    }

    if (
      project.network.existingRack === false &&
      project.network.rackRequired === null
    ) {
      addQuestion(questions, {
        key: "network.rackRequired",
        question:
          "Would you like the new equipment organized in a dedicated wall cabinet or floor rack?",
        reason:
          "A rack may be required to safely organize switches, recorders, patch panels, power, and cable management.",
        priority: "normal",
        category: "network",
      });
    }
  }

  /*
   * ---------------------------------------------------------
   * Wi-Fi scope
   * ---------------------------------------------------------
   */

  if (project.wifi.requested) {
    if (project.wifi.coverageGoals.length === 0) {
      addQuestion(questions, {
        key: "wifi.coverageGoals",
        question:
          "Which indoor and outdoor areas need reliable Wi-Fi coverage?",
        reason:
          "Coverage zones determine access-point quantity, placement, cabling, and outdoor equipment needs.",
        priority: "critical",
        category: "wifi",
      });
    }

    if (!hasKnownQuantity(project.wifi.estimatedConcurrentUsers)) {
      addQuestion(questions, {
        key: "wifi.estimatedConcurrentUsers",
        question:
          "Approximately how many people and connected devices may use the Wi-Fi at the same time?",
        reason:
          "Concurrent device count affects access-point density, switching, bandwidth, and system capacity.",
        priority: "high",
        category: "wifi",
      });
    }

    if (project.wifi.guestNetworkRequired === null) {
      addQuestion(questions, {
        key: "wifi.guestNetworkRequired",
        question:
          "Do you need a separate guest Wi-Fi network from the business or private network?",
        reason:
          "Guest networking may require VLANs, firewall rules, captive portals, and additional configuration.",
        priority: "normal",
        category: "wifi",
      });
    }

    if (
      project.wifi.indoorCoverage === null ||
      project.wifi.outdoorCoverage === null
    ) {
      addQuestion(questions, {
        key: "wifi.coverageType",
        question:
          "Should the Wi-Fi cover indoor areas, outdoor areas, or both?",
        reason:
          "Outdoor coverage requires weather-rated access points and different placement assumptions.",
        priority: "high",
        category: "wifi",
      });
    }
  }

  /*
   * ---------------------------------------------------------
   * Access-control scope
   * ---------------------------------------------------------
   */

  if (project.accessControl.requested) {
    if (!hasKnownQuantity(project.accessControl.controlledDoorCount)) {
      addQuestion(questions, {
        key: "accessControl.controlledDoorCount",
        question:
          "How many doors should receive electronic access control?",
        reason:
          "Door count is the primary driver for readers, locks, controllers, power supplies, cabling, and labor.",
        priority: "critical",
        category: "access_control",
      });
    }

    if (project.accessControl.credentialTypes.length === 0) {
      addQuestion(questions, {
        key: "accessControl.credentialTypes",
        question:
          "How should people unlock the doors: key cards, fobs, PIN codes, mobile credentials, or a combination?",
        reason:
          "Credential type affects reader selection, licensing, enrollment, and system configuration.",
        priority: "high",
        category: "access_control",
      });
    }

    if (project.accessControl.existingSystem === null) {
      addQuestion(questions, {
        key: "accessControl.existingSystem",
        question:
          "Is there an existing access-control, alarm, or intercom system that must be integrated or removed?",
        reason:
          "Existing systems can change compatibility, demolition, wiring, programming, and licensing requirements.",
        priority: "high",
        category: "access_control",
      });
    }

    if (project.accessControl.remoteManagementRequired === null) {
      addQuestion(questions, {
        key: "accessControl.remoteManagementRequired",
        question:
          "Do administrators need to manage users and doors remotely?",
        reason:
          "Remote management affects platform selection, subscriptions, network access, and configuration.",
        priority: "normal",
        category: "access_control",
      });
    }
  }

  /*
   * ---------------------------------------------------------
   * Cabling and installation conditions
   * ---------------------------------------------------------
   */

  if (project.cabling.existingCablingAvailable === null) {
    addQuestion(questions, {
      key: "cabling.existingCablingAvailable",
      question:
        "Is there existing network cabling that can be reused, or should we assume new cable runs?",
      reason:
        "Reusable cabling may reduce materials and labor, but it must be tested and verified.",
      priority: "high",
      category: "cabling",
    });
  }

  if (project.cabling.pathwayType.length === 0) {
    addQuestion(questions, {
      key: "cabling.pathwayType",
      question:
        "What cable pathways are available, such as drop ceilings, attic space, conduit, cable tray, open structure, or inside finished walls?",
      reason:
        "Cable pathways strongly influence labor hours, hardware, lift use, wall repair, and feasibility.",
      priority: "critical",
      category: "cabling",
    });
  }

  if (project.cabling.wiringStyle === "unknown") {
    addQuestion(questions, {
      key: "cabling.wiringStyle",
      question:
        "Should wiring be concealed, installed in exposed conduit, or use a mixture of both?",
      reason:
        "The desired finish level changes pathway materials, installation time, and appearance.",
      priority: "normal",
      category: "cabling",
    });
  }

  if (project.cabling.trenchingRequired === null) {
    addQuestion(questions, {
      key: "cabling.trenchingRequired",
      question:
        "Will any devices require underground trenching or cable runs between separate buildings, gates, poles, or detached structures?",
      reason:
        "Underground and interbuilding pathways can require trenching, permits, fiber, surge protection, and specialty labor.",
      priority: "high",
      category: "cabling",
    });
  }

  if (project.installation.afterHoursRequired === null) {
    addQuestion(questions, {
      key: "installation.afterHoursRequired",
      question:
        "Must any work be completed after hours, overnight, or during a restricted installation window?",
      reason:
        "Restricted work windows affect labor rates, staffing, duration, and scheduling.",
      priority: "normal",
      category: "installation",
    });
  }

  if (project.installation.ladderAccessPossible === null) {
    addQuestion(questions, {
      key: "installation.ladderAccessPossible",
      question:
        "Can the installation areas be safely reached by ladder, or will a scissor lift, boom lift, or other access equipment be required?",
      reason:
        "Access equipment affects rental cost, delivery, safety planning, and labor.",
      priority: "high",
      category: "installation",
    });
  }

  /*
   * ---------------------------------------------------------
   * Commercial and planning questions
   * ---------------------------------------------------------
   */

  const isCommercial =
    project.property.projectType !== null &&
    project.property.projectType !== "residential";

  if (isCommercial && project.customerIntent.futureExpansion === null) {
    addQuestion(questions, {
      key: "customerIntent.futureExpansion",
      question:
        "Should the system be designed for future expansion, additional locations, or more devices later?",
      reason:
        "Future growth affects switch capacity, rack size, licensing, controller size, storage, and spare cabling.",
      priority: "normal",
      category: "commercial",
    });
  }

  return questions.sort((a, b) => {
    const priorityDifference =
      priorityOrder[a.priority] - priorityOrder[b.priority];

    if (priorityDifference !== 0) {
      return priorityDifference;
    }

    return a.key.localeCompare(b.key);
  });
}

export function getNextEstimatorQuestion(
  project: ProjectEstimate
): EstimatorQuestion | null {
  const questions = getEstimatorQuestions(project);

  return questions[0] ?? null;
}

export function getUnansweredEstimatorQuestionKeys(
  project: ProjectEstimate
): string[] {
  return getEstimatorQuestions(project).map((question) => question.key);
}

export function isProjectReadyForPreliminaryPricing(
  project: ProjectEstimate
): boolean {
  const unansweredQuestions = getEstimatorQuestions(project);

  return !unansweredQuestions.some(
    (question) =>
      question.priority === "critical" ||
      question.priority === "high"
  );

import type { ProjectEstimate } from "../domain/project-estimate";

import type { EstimatorQuestion } from "./question-planner";
import { getEstimatorQuestions } from "./question-planner";

import { selectApplicableEstimatorPlaybooks } from "../knowledge/playbook-registry";
import { resolveEstimatorPlaybook } from "../knowledge/playbook-resolver";
import type { PlaybookQuestion } from "../knowledge/playbook";

export type CombinedEstimatorQuestion = EstimatorQuestion & {
  source: "core" | "playbook";
  playbookId: string | null;

  answerType:
    | "text"
    | "boolean"
    | "number"
    | "single_choice"
    | "multiple_choice";

  choices: string[];
  promptGuidance: string;
  ruleTags: string[];

  requiredForPreliminaryEstimate: boolean;
  requiredForFinalQuote: boolean;
};

const priorityOrder: Record<
  CombinedEstimatorQuestion["priority"],
  number
> = {
  critical: 0,
  high: 1,
  normal: 2,
  optional: 3,
};

const categoryOrder: Record<
  CombinedEstimatorQuestion["category"],
  number
> = {
  project: 0,
  property: 1,
  cameras: 2,
  wifi: 3,
  access_control: 4,
  network: 5,
  cabling: 6,
  installation: 7,
  commercial: 8,
};

function mapCoreQuestion(
  question: EstimatorQuestion
): CombinedEstimatorQuestion {
  return {
    ...question,

    source: "core",
    playbookId: null,

    answerType: inferCoreAnswerType(question.key),

    choices: [],

    promptGuidance: "",

    ruleTags: [],

    requiredForPreliminaryEstimate:
      question.priority === "critical" ||
      question.priority === "high",

    requiredForFinalQuote: true,
  };
}

function inferCoreAnswerType(
  key: string
): CombinedEstimatorQuestion["answerType"] {
  const booleanKeys = new Set([
    "property.occupiedDuringInstall",
    "cameras.existingSystem",
    "cameras.remoteViewingRequired",
    "network.existingRouter",
    "network.existingSwitches",
    "network.existingRack",
    "network.rackRequired",
    "wifi.guestNetworkRequired",
    "accessControl.existingSystem",
    "accessControl.remoteManagementRequired",
    "cabling.existingCablingAvailable",
    "cabling.trenchingRequired",
    "installation.afterHoursRequired",
    "installation.ladderAccessPossible",
  ]);

  const numberKeys = new Set([
    "property.squareFootage",
    "property.numberOfFloors",
    "property.ceilingHeightFeet",
    "cameras.recordingDays",
    "wifi.estimatedConcurrentUsers",
    "accessControl.controlledDoorCount",
  ]);

  if (booleanKeys.has(key)) {
    return "boolean";
  }

  if (numberKeys.has(key)) {
    return "number";
  }

  return "text";
}

function mapPlaybookCategory(
  category: PlaybookQuestion["category"]
): EstimatorQuestion["category"] {
  switch (category) {
    case "discovery":
      return "project";

    case "property":
      return "property";

    case "cameras":
      return "cameras";

    case "network":
      return "network";

    case "wifi":
      return "wifi";

    case "access_control":
      return "access_control";

    case "cabling":
      return "cabling";

    case "installation":
    case "safety":
      return "installation";

    case "commercial":
    case "pricing":
    case "audio_visual":
      return "commercial";

    default:
      return "project";
  }
}

function mapPlaybookQuestion(
  question: PlaybookQuestion,
  playbookId: string
): CombinedEstimatorQuestion {
  return {
    key: question.id,

    question: question.question,

    reason: question.reason,

    priority: question.priority,

    category: mapPlaybookCategory(question.category),

    source: "playbook",

    playbookId,

    answerType: question.answerType,

    choices: question.choices,

    promptGuidance: question.promptGuidance,

    ruleTags: question.ruleTags,

    requiredForPreliminaryEstimate:
      question.requiredForPreliminaryEstimate,

    requiredForFinalQuote:
      question.requiredForFinalQuote,
  };
}

function removeDuplicateQuestions(
  questions: CombinedEstimatorQuestion[]
): CombinedEstimatorQuestion[] {
  const seenKeys = new Set<string>();

  return questions.filter((question) => {
    if (seenKeys.has(question.key)) {
      return false;
    }

    seenKeys.add(question.key);

    return true;
  });
}

export function getCombinedEstimatorQuestions(
  project: ProjectEstimate
): CombinedEstimatorQuestion[] {
  const coreQuestions =
    getEstimatorQuestions(project).map(
      mapCoreQuestion
    );

  const playbookQuestions =
    selectApplicableEstimatorPlaybooks(project)
      .flatMap((playbook) =>
        resolveEstimatorPlaybook(
          project,
          playbook
        ).questions.map((question) =>
          mapPlaybookQuestion(
            question,
            playbook.id
          )
        )
      );

  const combined = removeDuplicateQuestions([
    ...coreQuestions,
    ...playbookQuestions,
  ]);

  return combined.sort((a, b) => {
    const priorityDifference =
      priorityOrder[a.priority] -
      priorityOrder[b.priority];

    if (priorityDifference !== 0) {
      return priorityDifference;
    }

    const categoryDifference =
      categoryOrder[a.category] -
      categoryOrder[b.category];

    if (categoryDifference !== 0) {
      return categoryDifference;
    }

    if (a.source !== b.source) {
      return a.source === "playbook"
        ? -1
        : 1;
    }

    return a.key.localeCompare(b.key);
  });
}

export function getNextCombinedEstimatorQuestion(
  project: ProjectEstimate
): CombinedEstimatorQuestion | null {
  return (
    getCombinedEstimatorQuestions(project)[0] ??
    null
  );
}

export function isCombinedProjectReadyForPreliminaryPricing(
  project: ProjectEstimate
): boolean {
  const questions =
    getCombinedEstimatorQuestions(project);

  return !questions.some(
    (question) =>
      question.requiredForPreliminaryEstimate
  );
}

import type { ProjectEstimate } from "../domain/project-estimate";

import {
  getCombinedEstimatorQuestions,
  getNextCombinedEstimatorQuestion,
  isCombinedProjectReadyForPreliminaryPricing,
} from "./combined-question-planner";

import {
  createEstimatorConversationState,
  type EstimatorConversationState,
} from "./conversation-state";

export type ConversationTurnResult = {
  project: ProjectEstimate;
  conversation: EstimatorConversationState;
};

type StartConversationOptions = {
  sessionId: string;
  project: ProjectEstimate;
};

export function startEstimatorConversation({
  sessionId,
  project,
}: StartConversationOptions): ConversationTurnResult {
  const now = new Date().toISOString();

  const conversation =
    createEstimatorConversationState(sessionId);

  const nextQuestion =
    getNextCombinedEstimatorQuestion(project);

  const unansweredQuestions =
    getCombinedEstimatorQuestions(project);

  const readyForPricing =
    isCombinedProjectReadyForPreliminaryPricing(
      project
    );

  if (!nextQuestion) {
    conversation.status = readyForPricing
      ? "ready_for_pricing"
      : "completed";

    conversation.readyForPricing =
      readyForPricing;

    conversation.confidenceScore =
      calculateConversationConfidence(project);

    conversation.lastAssistantMessage =
      "I have enough information to prepare a preliminary SmartNET estimate.";

    conversation.messages.push({
      id: createMessageId(),
      role: "assistant",
      content:
        conversation.lastAssistantMessage,
      createdAt: now,
    });

    conversation.updatedAt = now;

    return {
      project,
      conversation,
    };
  }

  conversation.status = "awaiting_customer";

  conversation.currentQuestion =
    nextQuestion.question;

  conversation.currentQuestionKey =
    nextQuestion.key;

  conversation.unansweredQuestionKeys =
    unansweredQuestions.map(
      (question) => question.key
    );

  conversation.completedQuestionKeys = [];

  conversation.readyForPricing =
    readyForPricing;

  conversation.confidenceScore =
    calculateConversationConfidence(project);

  conversation.lastAssistantMessage =
    nextQuestion.question;

  conversation.messages.push({
    id: createMessageId(),
    role: "assistant",
    content: nextQuestion.question,
    createdAt: now,
  });

  conversation.updatedAt = now;

  return {
    project,
    conversation,
  };
}

export function refreshEstimatorConversation(
  project: ProjectEstimate,
  conversation: EstimatorConversationState
): ConversationTurnResult {
  const now = new Date().toISOString();

  const previousQuestionKeys = new Set(
    conversation.unansweredQuestionKeys
  );

  const unansweredQuestions =
    getCombinedEstimatorQuestions(project);

  const unansweredQuestionKeys =
    unansweredQuestions.map(
      (question) => question.key
    );

  const completedQuestionKeys =
    conversation.completedQuestionKeys.slice();

  for (const previousKey of previousQuestionKeys) {
    if (
      !unansweredQuestionKeys.includes(
        previousKey
      ) &&
      !completedQuestionKeys.includes(
        previousKey
      )
    ) {
      completedQuestionKeys.push(previousKey);
    }
  }

  const nextQuestion =
    getNextCombinedEstimatorQuestion(project);

  const readyForPricing =
    isCombinedProjectReadyForPreliminaryPricing(
      project
    );

  const finalAssistantMessage =
    nextQuestion?.question ??
    "I have enough information to prepare a preliminary SmartNET estimate.";

  const nextConversation: EstimatorConversationState =
    {
      ...conversation,

      status: nextQuestion
        ? "awaiting_customer"
        : readyForPricing
          ? "ready_for_pricing"
          : "completed",

      currentQuestion:
        nextQuestion?.question ?? null,

      currentQuestionKey:
        nextQuestion?.key ?? null,

      unansweredQuestionKeys,

      completedQuestionKeys,

      lastAssistantMessage:
        finalAssistantMessage,

      readyForPricing,

      confidenceScore:
        calculateConversationConfidence(
          project
        ),

      updatedAt: now,
    };

  if (
    finalAssistantMessage !==
    conversation.lastAssistantMessage
  ) {
    nextConversation.messages = [
      ...conversation.messages,
      {
        id: createMessageId(),
        role: "assistant",
        content: finalAssistantMessage,
        createdAt: now,
      },
    ];
  }

  return {
    project,
    conversation: nextConversation,
  };
}

function calculateConversationConfidence(
  project: ProjectEstimate
): number {
  const questions =
    getCombinedEstimatorQuestions(project);

  const remainingCritical =
    questions.filter(
      (question) =>
        question.priority === "critical"
    ).length;

  const remainingHigh =
    questions.filter(
      (question) =>
        question.priority === "high"
    ).length;

  const remainingNormal =
    questions.filter(
      (question) =>
        question.priority === "normal"
    ).length;

  const penalty =
    remainingCritical * 12 +
    remainingHigh * 6 +
    remainingNormal * 2;

  return Math.max(
    0,
    Math.min(100, 100 - penalty)
  );
}

function createMessageId(): string {
  return crypto.randomUUID();
}
